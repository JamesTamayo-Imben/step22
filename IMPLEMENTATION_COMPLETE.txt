🔗 API ENDPOINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

POST /api/password-reset/send-otp
  ├─ Input: { email }
  ├─ Process: Generate OTP → Send email → Cache
  └─ Response: { success, message }

POST /api/password-reset/verify-otp
  ├─ Input: { email, otp }
  ├─ Process: Validate OTP → Create token → Cache
  └─ Response: { success, message, verification_token }

POST /api/password-reset/reset
  ├─ Input: { email, otp, password, password_confirmation }
  ├─ Process: Validate token → Hash password → Update user
  └─ Response: { success, message }


🔒 SECURITY FEATURES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ OTP Security
   • Random 6-digit generation
   • 10-minute expiration
   • Cached (not in database)
   • Unique per email

✅ Password Security
   • Minimum 8 characters
   • Bcrypt hashing
   • Confirmation validation
   • No hints sent via email

✅ Token Security
   • Cryptographically random
   • 15-minute expiration
   • Separate from OTP
   • Cannot be reused

✅ Privacy Protection
   • Email enumeration prevention
   • Success returned even if email not found
   • Detailed errors only to legitimate users
   • Comprehensive audit logging

🧪 TESTING COMMANDS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

# 1. Request OTP
curl -X POST http://localhost:8000/api/password-reset/send-otp \
  -H "Content-Type: application/json" \
  -d '{"email": "test@kld.edu.ph"}'

# 2. Verify OTP (use code from email)
curl -X POST http://localhost:8000/api/password-reset/verify-otp \
  -H "Content-Type: application/json" \
  -d '{"email": "test@kld.edu.ph", "otp": "123456"}'

# 3. Reset Password
curl -X POST http://localhost:8000/api/password-reset/reset \
  -H "Content-Type: application/json" \
  -d '{
    "email": "test@kld.edu.ph",
    "otp": "123456",
    "password": "NewPassword123",
    "password_confirmation": "NewPassword123"
  }'
