  ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

IMPACT:
  - Old rating_score column removed
  - Three new nullable integer columns added
  - Only satisfaction_rating is required (1-5)
  - completeness_rating and engagement_rating are optional (can be NULL)

5. CONTROLLER UPDATE
================================================================================

FILE: app/Http/Controllers/User/UserProjectController.php
LOCATION: upsertRating() method

BEFORE VALIDATION:
  $validated = $request->validate([
    'rating' => ['required', 'integer', 'min:1', 'max:5'],
    'comment' => ['nullable', 'string', 'max:1000'],
  ]);

AFTER VALIDATION:
  $validated = $request->validate([
    'satisfaction_rating' => ['required', 'integer', 'min:1', 'max:5'],
    'completeness_rating' => ['nullable', 'integer', 'min:1', 'max:5'],
    'engagement_rating' => ['nullable', 'integer', 'min:1', 'max:5'],
    'comment' => ['nullable', 'string', 'max:1000'],
  ]);

VALIDATION RULES EXPLAINED:
  satisfaction_rating:
    - required: User must submit at least this rating
    - integer: Must be whole number
    - min:1, max:5: Must be between 1-5 stars
  
  completeness_rating:
    - nullable: Can be empty/null (optional)
    - integer: If provided, must be whole number
    - min:1, max:5: If provided, must be between 1-5
  
  engagement_rating:
    - Same as completeness_rating (optional)
  
  comment:
    - nullable: Can be empty
    - string: Must be text
    - max:1000: Maximum 1000 characters

BEFORE STORAGE:
  $rating->satisfaction_rating = $validated['rating'];

AFTER STORAGE:
  $rating->satisfaction_rating = $validated['satisfaction_rating'];
  $rating->completeness_rating = $validated['completeness_rating'] ?? null;
  $rating->engagement_rating = $validated['engagement_rating'] ?? null;

IMPACT:
  - Stores all three ratings in database
  - If completeness/engagement not provided, stores NULL
  - One-rating-per-user-per-project validation still applies

================================================================================
                         BACKWARDS COMPATIBILITY
================================================================================

If old code sends only "rating" field:
  - Will fail validation (satisfaction_rating required)
  - Returns HTTP 422 error
  - Need to update all old API calls

If database has old rating_score column data:
  - Migration doesn't automatically migrate old data
  - Need to run manual data migration if needed:
    
    UPDATE ratings 
    SET satisfaction_rating = rating_score 
    WHERE satisfaction_rating IS NULL;
    
    Then drop old column:
    ALTER TABLE ratings DROP COLUMN rating_score;
